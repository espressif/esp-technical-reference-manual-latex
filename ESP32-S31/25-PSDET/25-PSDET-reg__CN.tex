
\begin{register}{H}{LP\_ANA\_BOD\_MODE0\_CNTL\_REG}{0x0000}\label{regdesc:LPANABODMODE0CNTLREG}\regfield{LP\_ANA\_BOD\_MODE0\_RESET\_ENA}{1}{31}{{0}}%
\regfield{LP\_ANA\_BOD\_MODE0\_RESET\_SEL}{1}{30}{{0}}%
\regfield{LP\_ANA\_BOD\_MODE0\_INTR\_ENA}{1}{29}{{0}}%
\regfield{LP\_ANA\_BOD\_MODE0\_CNT\_CLR}{1}{28}{{0}}%
\regfield{LP\_ANA\_BOD\_MODE0\_RESET\_WAIT}{10}{18}{{0x3ff}}%
\regfield{LP\_ANA\_BOD\_MODE0\_INTR\_WAIT}{10}{8}{{1}}%
\regfield{LP\_ANA\_BOD\_MODE0\_PD\_RF\_ENA}{1}{7}{{0}}%
\regfield{LP\_ANA\_BOD\_MODE0\_CLOSE\_FLASH\_ENA}{1}{6}{{0}}%
\regfield{(reserved)}{6}{0}{000000}%
\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}
\label{fielddesc:LPANABODMODE0CLOSEFLASHENA}\item [LP\_ANA\_BOD\_MODE0\_CLOSE\_FLASH\_ENA] 配置是否使能欠压检测器触发 flash 挂起。\\
0：禁用\\
1：使能\\
(R/W)
\label{fielddesc:LPANABODMODE0PDRFENA}\item [LP\_ANA\_BOD\_MODE0\_PD\_RF\_ENA]  配置是否使能欠压检测器关闭射频模块。\\
0：禁用\\
1：使能\\
(R/W)
\label{fielddesc:LPANABODMODE0INTRWAIT}\item [LP\_ANA\_BOD\_MODE0\_INTR\_WAIT] 配置欠压信号有效后，产生中断的时间，单位是 LP\_FAST\_CLK 周期。(R/W)
\label{fielddesc:LPANABODMODE0RESETWAIT}\item [LP\_ANA\_BOD\_MODE0\_RESET\_WAIT] 配置欠压信号有效后，产生复位的时间，单位是 LP\_FAST\_CLK 周期。(R/W)
\label{fielddesc:LPANABODMODE0CNTCLR}\item [LP\_ANA\_BOD\_MODE0\_CNT\_CLR] 配置是否清空欠压检测器计数器。\\
0：不清空\\
1：清空\\
(R/W)
\label{fielddesc:LPANABODMODE0INTRENA}\item [LP\_ANA\_BOD\_MODE0\_INTR\_ENA] 配置是否使能欠压检测器模式 0 的中断。此域为 1 时 \hyperref[fielddesc:LPANABODMODE0INTRAW]{LP\_ANA\_BOD\_MODE0\_INT\_RAW} 和 \hyperref[fielddesc:LPANABODMODE0LPINTRAW]{LP\_ANA\_BOD\_MODE0\_LP\_INT\_RAW} 才有效。\\
0：禁用\\
1：使能\\
(R/W)
\label{fielddesc:LPANABODMODE0RESETSEL}\item [LP\_ANA\_BOD\_MODE0\_RESET\_SEL] 配置欠压检测器被触发后的复位方式。\\
0：芯片复位\\
1：系统复位\\
(R/W)
\label{fielddesc:LPANABODMODE0RESETENA}\item [LP\_ANA\_BOD\_MODE0\_RESET\_ENA] 配置是否使能欠压检测器复位。\\
0：禁用\\
1：使能\\
(R/W)
\end{reglist}\end{regdesc}
\end{register}
