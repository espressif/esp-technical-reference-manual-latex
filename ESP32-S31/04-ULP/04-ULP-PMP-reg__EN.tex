The PMP configuration register format for any PMP entry is shown as follows.

\begin{register}{H}{pmpXcfg}{}
\label{regdesc:lppmpXcfg}%
\regfield{L}{1}{7}{{0}}%
\regfield{Reserved}{2}{5}{{0}}%
\regfield{A}{2}{3}{{0}}%
\regfield{X}{1}{2}{{0}}%
\regfield{W}{1}{1}{{0}}%
\regfield{R}{1}{0}{{0}}%
\reglabel{Reset}\regnewline%
\begin{regdesc}
\begin{reglist}
\label{fielddesc:LPPMPXCFGL}\item [L] Configures lock bit. Once set, it can only be cleared through a core reset. (R/W)
\label{fielddesc:LPPMPXCFGA}\item [A] Configures the address matching mode. \\
0: Off \\
1: TOR (top of range)\\
2: Not supported \\
3: NAPOT (naturally aligned power-of-two region >= 128 bytes)\\
(R/W)
\label{fielddesc:LPPMPXCFGX}\item [X] Configures execute permissions. After it is set, execution from the address range is permitted. (R/W)
\label{fielddesc:LPPMPXCFGW}\item [W] Configures write permissions. Once set, memory writes to the address range are permitted. (R/W)
\label{fielddesc:LPPMPXCFGR}\item [R] Configures read permissions. Once set, memory can be read from the address range. (R/W)
\end{reglist}
\end{regdesc}
\end{register}
