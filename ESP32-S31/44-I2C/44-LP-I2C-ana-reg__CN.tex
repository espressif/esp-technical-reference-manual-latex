\begin{register}{H}{LP\_I2C\_ANA\_MST\_I2C0\_CTRL\_REG}{0x0000}\label{regdesc:LPI2CANAMSTI2C0CTRLREG}
\regfield{(reserved)}{6}{26}{000000}%
\regfield{LP\_I2C\_ANA\_MST\_I2C0\_BUSY}{1}{25}{{0}}%
\regfield{LP\_I2C\_ANA\_MST\_I2C0\_CTRL}{25}{0}{{0x00000}}%
\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}
\label{fielddesc:LPI2CANAMSTI2C0CTRL}\item [LP\_I2C\_ANA\_MST\_I2C0\_CTRL] 配置 I2C0 传输信息。\\
bit[24]：配置读写操作\\
0：读\\
1：写\\
bit[23:16]：配置第二个数据包\\
bit[15:8]：配置首个数据包\\
bit[6:0]：配置 I2C 地址\\
该寄存器配置完成后 I2C0 主机将产生 I2C 读写信号。\\
(R/W)
\label{fielddesc:LPI2CANAMSTI2C0BUSY}\item [LP\_I2C\_ANA\_MST\_I2C0\_BUSY] 表示 I2C0 是否正在传输数据。\\
0：I2C0 未进行数据传输\\
1：I2C0 正在传输数据\\
(RO)
\end{reglist}\end{regdesc}
\end{register}
