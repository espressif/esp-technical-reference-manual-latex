\begin{register}{H}{LP\_I2C\_ANA\_MST\_I2C0\_CTRL\_REG}{0x0000}\label{regdesc:LPI2CANAMSTI2C0CTRLREG}
\regfield{(reserved)}{6}{26}{000000}%
\regfield{LP\_I2C\_ANA\_MST\_I2C0\_BUSY}{1}{25}{{0}}%
\regfield{LP\_I2C\_ANA\_MST\_I2C0\_CTRL}{25}{0}{{0x00000}}%
\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}
\label{fielddesc:LPI2CANAMSTI2C0CTRL}\item [LP\_I2C\_ANA\_MST\_I2C0\_CTRL] Configures the transmission information for I2C0. \\
Bit[24]: Configures the read or write operation\\
0: Read \\
1: Write \\
Bit[23:16]: Configures the second data packet\\
Bit[15:8]: Configures the first data packet\\
Bit[6:0]: Configures the I2C address\\
Once this register is configured, the I2C0 master will generate I2C read or write signals.\\
(R/W)
\label{fielddesc:LPI2CANAMSTI2C0BUSY}\item [LP\_I2C\_ANA\_MST\_I2C0\_BUSY] Represents whether I2C0 is currently transferring data. \\
0: I2C0 is not transferring data\\
1: I2C0 is transferring data \\
(RO)
\end{reglist}\end{regdesc}
\end{register}
