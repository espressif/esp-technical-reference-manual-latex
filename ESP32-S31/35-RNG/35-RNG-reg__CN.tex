\begin{register}{H}{TRNG\_CONF\_REG}{0x2081\_4000}\label{regdesc:TRNGCONFREG}
\regfield{TRNG\_SAMPLE\_ENABLE}{1}{31}{{0}}%
\regfield{TRNG\_NOISE\_CRC\_EN}{1}{30}{{0x1}}%
\regfield{reserved}{1}{29}{{0x0}}%
\regfield{TRNG\_FIFO\_RESET}{1}{28}{{0x0}}%
\regfield{TRNG\_ADAPTIVE\_VALUE\_C}{9}{19}{{0x00}}%
\regfield{TRNG\_REPETITION\_VALUE\_C}{9}{10}{{0x00}}%
\regfield{TRNG\_NOISE\_POS\_SEL}{5}{5}{{0x00}}%
\regfield{TRNG\_NOISE\_SOURCE\_SEL}{5}{0}{{0x00}}%

\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}

\label{fielddesc:TRNGSAMPLEENABLE}\item [TRNG\_SAMPLE\_ENABLE] 配置是否使能 BUF\_CHAIN 环形振荡器噪声源。\\
0：关闭\\
1：使能\\
(R/W)
\label{fielddesc:TRNGNOISECRCEN}\item [TRNG\_NOISE\_CRC\_EN] 配置是否使能 CRC 随机数及相关噪声源。\\
0：关闭\\
1：使能\\
(R/W)


\label{fielddesc:TRNGFIFORESET}\item [TRNG\_FIFO\_RESET] 配置是否复位并清空 RNG 内部用于存储随机数的 FIFO。\\
0：无效，无作用\\
1：复位\\
(WT)

\label{fielddesc:TRNGADAPTIVEVALUEC}\item [TRNG\_ADAPTIVE\_VALUE\_C] 配置 Adaptive Test（自适应测试）的阈值。\\
(R/W)


\label{fielddesc:TRNGREPETITIONVALUEC}\item [TRNG\_REPETITION\_VALUE\_C] 配置 Repetition Test（重复计数测试）的阈值。\\
(R/W)

\item [见下页...]
\end{reglist}\end{regdesc}
\end{register}

\begin{register}{H}{TRNG\_CONF\_REG}{0x2081\_4000}
\begin{regdesc}\begin{reglist}
\item [接上页...]

\label{fielddesc:TRNGNOISEPOSSEL}\item [TRNG\_NOISE\_POS\_SEL] 配置 NIST 通路噪声采样位置选择。编码方式为独热；仅应将下列取值之一写入本域。\\
0x1：BUF\_CHAIN\\
0x2：RC\_FAST\_CLK\\
0x4：RF\_ADC\\
0x8：SAR\_ADC\\
0x10：CRC\_DATA\\
其他值：无效。可能会触发异常\\
(R/W)

\label{fielddesc:TRNGNOISESOURCESEL}\item [TRNG\_NOISE\_SOURCE\_SEL] 配置 NIST 通路各噪声源的输入使能（可多选）。\\
Bit 0 置 1：使能 BUF\_CHAIN\\
Bit 1 置 1：使能 RC\_FAST\_CLK\\
Bit 2 置 1：使能 RF\_ADC\\
Bit 3 置 1：使能 SAR\_ADC\\
Bit 4 置 1：使能 CRC\_DATA，选择此项时忽略其他噪声源输入\\
(R/W)
\end{reglist}\end{regdesc}
\end{register}


\begin{register}{H}{TRNG\_DEBUG\_CONF\_REG}{0x2081\_4004}\label{regdesc:TRNGDEBUGCONFREG}
\regfield{reserved}{1}{31}{{0x0}}%
\regfield{TRNG\_HEALTH\_TEST\_BYPASS}{1}{30}{{0}}%
\regfield{TRNG\_HEALTH\_TEST\_END}{1}{29}{{0}}%
\regfield{TRNG\_HEALTH\_TEST\_ERROR\_CODE}{3}{26}{{0}}%
\regfield{TRNG\_DEBUG\_MODE}{1}{25}{{0}}%
\regfield{TRNG\_DEBUG\_DATA}{8}{17}{{0}}%
\regfield{TRNG\_STARTUP\_TEST\_LIMIT}{16}{1}{{0x400}}%
\regfield{TRNG\_STARTUP\_TEST\_START}{1}{0}{{0}}%

\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}

\label{fielddesc:TRNGHEALTHTESTBYPASS}\item [TRNG\_HEALTH\_TEST\_BYPASS] 配置是否旁路健康测试。仅测试模式可将本寄存器置 1。\\
0：关闭旁路\\
1：旁路\\
(R/W)

\label{fielddesc:TRNGHEALTHTESTEND}\item [TRNG\_HEALTH\_TEST\_END] 配置是否立即结束当前健康测试。\\
0：无效，无作用\\
1：结束\\
(WT)

\label{fielddesc:TRNGHEALTHTESTERRORCODE}\item [TRNG\_HEALTH\_TEST\_ERROR\_CODE] 表示健康测试结果的错误代码。\\
0：无错误\\
1：熵源组合错误\\
2：Startup Test 未通过（由 Repetition Test 失败引起）\\
3：Startup Test 未通过（由 Adaptive Test 失败引起）\\
4：Repetition Test 未通过\\
5：Adaptive Test 未通过\\
(RO)

\label{fielddesc:TRNGDEBUGMODE}\item [TRNG\_DEBUG\_MODE] 配置是否进入调试模式。\\
0：正常工作模式\\
1：调试模式\\
(R/W)

\label{fielddesc:TRNGDEBUGDATA}\item [TRNG\_DEBUG\_DATA] 表示调试模式下捕获的随机熵源值。(R/W)

\label{fielddesc:TRNGSTARTUPTESTLIMIT}\item [TRNG\_STARTUP\_TEST\_LIMIT] 配置 Startup Test 中除固定 512 个采样之外的附加数据长度。\\
单位：采样个数\\
(R/W)
\label{fielddesc:TRNGSTARTUPTESTSTART}\item [TRNG\_STARTUP\_TEST\_START] 配置是否启动 Startup Test。\\
0：无效，无作用\\
1：启动\\
(WT)

\end{reglist}\end{regdesc}
\end{register}
