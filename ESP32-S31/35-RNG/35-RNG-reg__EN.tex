0: Invalid. No effect\\
1: Reset\\
(WT)

\label{fielddesc:TRNGADAPTIVEVALUEC}\item [TRNG\_ADAPTIVE\_VALUE\_C] Configures the threshold of Adaptive Test. (R/W)


\label{fielddesc:TRNGREPETITIONVALUEC}\item [TRNG\_REPETITION\_VALUE\_C] Configures the threshold of Repetition Test. (R/W)

\begin{register}{H}{TRNG\_CONF\_REG}{0x2081\_4000}
\begin{regdesc}\begin{reglist}
\item [Continued from the previous page...]

\label{fielddesc:TRNGNOISEPOSSEL}\item [TRNG\_NOISE\_POS\_SEL] Configures the noise sampling position selection in the NIST path. This field uses one-hot encoding, and only one of the following values should be written:\\
0x1: BUF\_CHAIN\\
0x2: RC\_FAST\_CLK\\
0x4: RF\_ADC\\
0x8: SAR\_ADC\\
0x10: CRC\_DATA\\
Other values: Invalid. Might trigger unexpected behaviors\\
(R/W)


\label{fielddesc:TRNGNOISESOURCESEL}\item [TRNG\_NOISE\_SOURCE\_SEL] Configures input enables of noise sources in the NIST path (multiple selections are allowed).\\
Bit 0 set to 1: Enable BUF\_CHAIN\\
Bit 1 set to 1: Enable RC\_FAST\_CLK\\
Bit 2 set to 1: Enable RF\_ADC\\
Bit 3 set to 1: Enable SAR\_ADC\\
Bit 4 set to 1: Enable CRC\_DATA. If selected, other noise source inputs are ignored\\
(R/W)
\end{reglist}\end{regdesc}
\end{register}


\begin{register}{H}{TRNG\_DEBUG\_CONF\_REG}{0x2081\_4004}\label{regdesc:TRNGDEBUGCONFREG}
\regfield{reserved}{1}{31}{{0x0}}%
\regfield{TRNG\_HEALTH\_TEST\_BYPASS}{1}{30}{{0}}%
\regfield{TRNG\_HEALTH\_TEST\_END}{1}{29}{{0}}%
\regfield{TRNG\_HEALTH\_TEST\_ERROR\_CODE}{3}{26}{{0}}%
\regfield{TRNG\_DEBUG\_MODE}{1}{25}{{0}}%
\regfield{TRNG\_DEBUG\_DATA}{8}{17}{{0}}%
\regfield{TRNG\_STARTUP\_TEST\_LIMIT}{16}{1}{{0x400}}%
\regfield{TRNG\_STARTUP\_TEST\_START}{1}{0}{{0}}%

\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}

\label{fielddesc:TRNGHEALTHTESTBYPASS}\item [TRNG\_HEALTH\_TEST\_BYPASS] Configures whether to bypass health tests. Only in test mode shall this register be set to 1.\\
0: Disable bypass\\
1: Bypass\\
(R/W)

\label{fielddesc:TRNGHEALTHTESTEND}\item [TRNG\_HEALTH\_TEST\_END] Configures whether to end the current health test immediately.\\
0: Invalid. No effect\\
1: End\\
(WT)

\label{fielddesc:TRNGHEALTHTESTERRORCODE}\item [TRNG\_HEALTH\_TEST\_ERROR\_CODE] Represents the health-test error code.\\
0: No error\\
1: Entropy source combination error\\
2: Startup Test failed (caused by Repetition Test failure)\\
3: Startup Test failed (caused by Adaptive Test failure)\\
4: Repetition Test failed\\
5: Adaptive Test failed\\
(RO)

\label{fielddesc:TRNGDEBUGMODE}\item [TRNG\_DEBUG\_MODE] Configures whether to enter debug mode.\\
0: Normal operating mode\\
1: Debug mode\\
(R/W)

\label{fielddesc:TRNGDEBUGDATA}\item [TRNG\_DEBUG\_DATA] Represents the random entropy-source value captured in debug mode. (R/W)

\label{fielddesc:TRNGSTARTUPTESTLIMIT}\item [TRNG\_STARTUP\_TEST\_LIMIT] Configures the additional data length in Startup Test, beyond the fixed 512 samples.\\
Measurement unit: Number of samples\\
(R/W)
\label{fielddesc:TRNGSTARTUPTESTSTART}\item [TRNG\_STARTUP\_TEST\_START] Configures whether to start Startup Test.\\
0: Invalid. No effect\\
1: Start Startup Test\\
(WT)

\end{reglist}\end{regdesc}
\end{register}
