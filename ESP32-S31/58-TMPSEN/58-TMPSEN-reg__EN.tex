\begin{register}{H}{TSENS\_CTRL\_REG}{0x0000}\label{regdesc:TSENSCTRLREG}
\regfield{(reserved)}{8}{24}{00000000}%
\regfield{TSENS\_POWER\_UP\_FORCE}{1}{23}{{0}}%
\regfield{TSENS\_POWER\_UP}{1}{22}{{0}}%
\regfield{TSENS\_CLK\_DIV}{8}{14}{{0x6}}%
\regfield{TSENS\_IN\_INV}{1}{13}{{0}}%
\regfield{TSENS\_INT\_EN}{1}{12}{{1}}%

\regfield{TSENS\_READY}{1}{8}{{0}}%
\regfield{TSENS\_OUT}{8}{0}{{0x0}}%
\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}
\label{fielddesc:TSENSOUT}\item [TSENS\_OUT] Represents the output data of the temperature sensor. (RO)
\label{fielddesc:TSENSREADY}\item [TSENS\_READY] Represents whether the temperature sensor output data is ready.\\
0: Not ready\\
1: Ready\\
(RO)
\label{fielddesc:TSENSSAMPLEEN}\item [TSENS\_SAMPLE\_EN] Configures whether to enable the sample signal for the wakeup module.\\
0: Disable\\
1: Enable\\
(R/W)
\label{fielddesc:TSENSWAKEUPMASK}\item [TSENS\_WAKEUP\_MASK] Configures whether to mask the wakeup signal.\\
0: Not mask\\
1: Mask\\
(R/W)
\label{fielddesc:TSENSINTEN}\item [TSENS\_INT\_EN] Configures whether to enable the temperature sensor interrupt.\\
0: Disable\\
1: Enable\\
(R/W)
\label{fielddesc:TSENSININV}\item [TSENS\_IN\_INV] Configures whether to invert the input data from the temperature sensor.\\
0: Not invert\\
1: Invert\\
(R/W)
\label{fielddesc:TSENSCLKDIV}\item [TSENS\_CLK\_DIV] Configures the clock divider of the temperature sensor clock. (R/W)
\label{fielddesc:TSENSPOWERUP}\item [TSENS\_POWER\_UP] Configures whether to power up the temperature sensor.\\
0: Power down\\
1: Power up\\
(R/W)
\label{fielddesc:TSENSPOWERUPFORCE}\item [TSENS\_POWER\_UP\_FORCE] Configures force control for the temperature sensor power-up.\\
0: Power-up controlled by power-up FSM\\
1: Power-up controlled by \hyperref[regdesc:TSENSCTRLREG]{TSENS\_CTRL\_REG}\\
(R/W)
\end{reglist}\end{regdesc}
\end{register}


\begin{register}{H}{TSENS\_CTRL2\_REG}{0x0004}\label{regdesc:TSENSCTRL2REG}
\regfield{(reserved)}{17}{15}{00000000000000000}%
\regfield{TSENS\_CLK\_INV}{1}{14}{{1}}%
\regfield{TSENS\_XPD\_FORCE}{2}{12}{{0x0}}%
\regfield{TSENS\_XPD\_WAIT}{12}{0}{{0x2}}%
\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}
\label{fielddesc:TSENSXPDWAIT}\item [TSENS\_XPD\_WAIT] Configures the number of clock cycles to wait after power up before releasing the sensor reset. (R/W)
\label{fielddesc:TSENSXPDFORCE}\item [TSENS\_XPD\_FORCE] Configures force control for the XPD\_TSENS signal.\\
Bit0:\\
0: Power down. Valid only when \hyperref[fielddesc:TSENSXPDFORCE]{TSENS\_XPD\_FORCE} bit1 is 1\\
1: Power up. Valid only when \hyperref[fielddesc:TSENSXPDFORCE]{TSENS\_XPD\_FORCE} bit1 is 1\\
Bit1:\\
0: XPD\_TSENS controlled by power-up FSM.\\
1: XPD\_TSENS forced to the value of \hyperref[fielddesc:TSENSXPDFORCE]{TSENS\_XPD\_FORCE} bit0.\\
(R/W)
\label{fielddesc:TSENSCLKINV}\item [TSENS\_CLK\_INV] Configures whether to invert the clock of the temperature sensor.\\
0: Not invert\\
1: Invert\\
(R/W)
\end{reglist}\end{regdesc}
\end{register}
