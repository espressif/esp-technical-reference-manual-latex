\begin{register}{H}{SPI\_MEM\_DPA\_CTRL\_REG}{0x0388}\label{regdesc:SPIMEMDPACTRLREG}
\regfield{(reserved)}{27}{5}{{0x0000000}}%
\regfield{SPI\_MEM\_CRYPT\_DPA\_SELECT\_REGISTER}{1}{4}{{0x0}}%
\regfield{SPI\_MEM\_CRYPT\_CALC\_D\_DPA\_EN}{1}{3}{{0x1}}%
\regfield{SPI\_MEM\_CRYPT\_SECURITY\_LEVEL}{3}{0}{{0x7}}%
\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}
\label{fielddesc:SPIMEMCRYPTDPASELECTREGISTER}\item [SPI\_MEM\_CRYPT\_DPA\_SELECT\_REGISTER] 配置由 eFuse 或寄存器控制抗 DPA 功能。\\
0：由 eFuse 控制抗 DPA 功能\\
1：由寄存器控制抗 DPA 功能\\
(R/W)
\label{fielddesc:SPIMEMCRYPTCALCDDPAEN}\item [SPI\_MEM\_CRYPT\_CALC\_D\_DPA\_EN] 配置是否在 XTS\_AES 算法中启用抗 DPA 功能。 \\
0：仅在计算 T 时启用抗 DPA 功能\\
1：计算 T 或是计算 D 时都启用抗 DPA 功能\\
请注意，此字段仅在 \hyperref[fielddesc:SPIMEMCRYPTSECURITYLEVEL]{SPI\_MEM\_CRYPT\_SECURITY\_LEVEL} 不为 0 时有效。\\
(R/W)
\label{fielddesc:SPIMEMCRYPTSECURITYLEVEL}\item [SPI\_MEM\_CRYPT\_SECURITY\_LEVEL] 配置外部加解密的安全等级。\\
0：抗 DPA 功能已关闭\\
1-7：数字越大，则加解密安全性更高\\
(R/W)
\end{reglist}\end{regdesc}
\end{register}
