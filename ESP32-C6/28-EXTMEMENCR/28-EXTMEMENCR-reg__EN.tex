\begin{regdesc}\begin{reglist}
\label{fielddesc:SPIMEMXTSDESTINATION}\item [SPI\_MEM\_XTS\_DESTINATION] Configures the type of external memory. Currently, it must be set to 0, as the Manual Encryption block only supports flash encryption. Errors may occur if users write 1.\\

(R/W)

\end{register}

\begin{register}{H}{SPI\_MEM\_DPA\_CTRL\_REG}{0x0388}\label{regdesc:SPIMEMDPACTRLREG}
\regfield{(reserved)}{27}{5}{{0x0000000}}%
\regfield{SPI\_MEM\_CRYPT\_DPA\_SELECT\_REGISTER}{1}{4}{{0x0}}%
\regfield{SPI\_MEM\_CRYPT\_CALC\_D\_DPA\_EN}{1}{3}{{0x1}}%
\regfield{SPI\_MEM\_CRYPT\_SECURITY\_LEVEL}{3}{0}{{0x7}}%
\reglabel{Reset}\regnewline%
\begin{regdesc}\begin{reglist}
\label{fielddesc:SPIMEMCRYPTDPASELECTREGISTER}\item [SPI\_MEM\_CRYPT\_DPA\_SELECT\_REGISTER] Configures whether the Anti-DPA function is controlled by eFuse or register. \\
0: The Anti-DPA function is configured by eFuse.\\
1: The Anti-DPA function is configured by register.\\
(R/W)
\label{fielddesc:SPIMEMCRYPTCALCDDPAEN}\item [SPI\_MEM\_CRYPT\_CALC\_D\_DPA\_EN] Configures whether to enable Anti-DPA in the XTS\_AES algorithm. \\
0: Enable Anti-DPA only when calculating T\\
1: Enable Anti-DPA both when calculating T and D\\
Note that this field is only effective when \hyperref[fielddesc:SPIMEMCRYPTSECURITYLEVEL]{SPI\_MEM\_CRYPT\_SECURITY\_LEVEL} is not 0.\\
(R/W)

\label{fielddesc:SPIMEMCRYPTSECURITYLEVEL}\item [SPI\_MEM\_CRYPT\_SECURITY\_LEVEL] Configures the security level of external memory encryption and decryption. \\
0: Disable the Anti-DPA function\\
1-7: The bigger the number is, the more secure the encryption and decryption are\\
(R/W)
\end{reglist}\end{regdesc}
\end{register}

\begin{register}{H}{SPI\_MEM\_XTS\_STATE\_REG}{0x0358}\label{regdesc:SPIMEMXTSSTATEREG}

\begin{regdesc}\begin{reglist}
